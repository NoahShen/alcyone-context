## 1. 复现方式与边界

将下方唯一 groovy 代码块完整保存至临时文件（如 `/tmp/run_probe.gradle`），在项目根目录执行：

```bash
./scripts/dev gradle --no-daemon -q -I /tmp/run_probe.gradle :integration-tests:vfs:runColdBackupProbe
```

仅操作脚本自己创建的两个临时目录，原库/文件删除也限于其中；不得换成真实数据目录。使用单次 Gradle 进程是因为脚本临时注册 JDBC Driver、设置线程 contextClassLoader，原脚本未恢复这些进程内状态，不宜加载到常驻 daemon。

探针是一次性取证材料，不是生产备份工具：反射调用 Kotlin ABI 对版本敏感；首个 Runtime 的异常路径未完整 finally 关闭、部分 Files.list 流未显式关闭、最终临时目录删除吞错。此次正常路径两次 Runtime 均关闭并完成断言；若未来改为正式工具，应先改用常规 Kotlin 调用并补齐失败清理，不能从本次成功推导异常清理已验证。

## 3. 完整脚本（开发方原文，未改动）

```groovy
allprojects {
    afterEvaluate { projectItem ->
        if (projectItem.path == ":integration-tests:vfs") {
            projectItem.tasks.register("runColdBackupProbe") {
                dependsOn(projectItem.tasks.named("testClasses"))
                doLast {
                    def classLoader = new URLClassLoader(
                        projectItem.sourceSets.test.runtimeClasspath.collect { it.toURI().toURL() } as URL[],
                        ClassLoader.systemClassLoader
                    )
                    Thread.currentThread().setContextClassLoader(classLoader)

                    // Register SQLite Driver with DriverManager
                    def driverClass = classLoader.loadClass("org.sqlite.JDBC")
                    def driverInstance = driverClass.getDeclaredConstructor().newInstance()
                    def registeredDriver = new java.sql.Driver() {
                        boolean acceptsURL(String u) { driverInstance.acceptsURL(u) }
                        java.sql.Connection connect(String u, Properties info) { driverInstance.connect(u, info) }
                        int getMajorVersion() { driverInstance.getMajorVersion() }
                        int getMinorVersion() { driverInstance.getMinorVersion() }
                        java.sql.DriverPropertyInfo[] getPropertyInfo(String u, Properties info) { driverInstance.getPropertyInfo(u, info) }
                        boolean jdbcCompliant() { driverInstance.jdbcCompliant() }
                        java.util.logging.Logger getParentLogger() { driverInstance.getParentLogger() }
                    }
                    java.sql.DriverManager.registerDriver(registeredDriver)

                    println("=== Alcyone VFS Cold Backup & Restore Probe ===")
                    def tempDir = java.nio.file.Files.createTempDirectory("alcyone-backup-probe-")
                    def stateDb = tempDir.resolve("state.db")
                    def docsDir = java.nio.file.Files.createDirectories(tempDir.resolve("docs"))
                    def backupDir = java.nio.file.Files.createTempDirectory("alcyone-backup-archive-")

                    try {
                        def vfsPathClass = classLoader.loadClass("com.github.noahshen.alcyone.context.vfs.VfsPath")
                        def vfsPath = vfsPathClass.Companion.parse("/resources/docs")

                        def localFsClass = classLoader.loadClass("com.github.noahshen.alcyone.context.vfs.runtime.MountBackend\$LocalFs")
                        def localBackend = localFsClass.getConstructor(java.nio.file.Path).newInstance(docsDir)

                        def mountConfigClass = classLoader.loadClass("com.github.noahshen.alcyone.context.vfs.runtime.MountConfig")
                        def mountConfig = mountConfigClass.getConstructor(vfsPathClass, String, classLoader.loadClass("com.github.noahshen.alcyone.context.vfs.runtime.MountBackend")).newInstance(vfsPath, "docs", localBackend)

                        def vfsLimitsClass = classLoader.loadClass("com.github.noahshen.alcyone.context.vfs.core.VfsLimits")
                        def vfsLimits = vfsLimitsClass.getConstructor(long, long).newInstance(16L * 1024 * 1024, 16L * 1024 * 1024)

                        def durationUnitClass = classLoader.loadClass("kotlin.time.DurationUnit")
                        def durationUnitSeconds = Enum.valueOf(durationUnitClass, "SECONDS")
                        def durationKtClass = classLoader.loadClass("kotlin.time.DurationKt")
                        def graceDuration = durationKtClass.toDuration(30, durationUnitSeconds)

                        def vfsRuntimeConfigClass = classLoader.loadClass("com.github.noahshen.alcyone.context.vfs.runtime.VfsRuntimeConfig")
                        def configCtor = vfsRuntimeConfigClass.getConstructors().find { it.parameterCount == 8 }
                        def config = configCtor.newInstance(
                            stateDb,
                            Collections.singleton("resources"),
                            Collections.singletonList(mountConfig),
                            vfsLimits,
                            null,
                            256,
                            graceDuration,
                            null
                        )

                        def vfsUriClass = classLoader.loadClass("com.github.noahshen.alcyone.context.vfs.VfsUri")
                        def testUri = vfsUriClass.Companion.parse("alcyone://resources/docs/test_report.txt")
                        def testContent = "Cold Backup Verification Content 2026-10-13".getBytes(java.nio.charset.StandardCharsets.UTF_8)
                        def testTags = new HashSet<>(Arrays.asList("audit", "backup-test"))
                        def nodeMetadataClass = classLoader.loadClass("com.github.noahshen.alcyone.context.vfs.NodeMetadata")
                        def metaCtor = nodeMetadataClass.getConstructors().find { it.parameterCount == 5 }
                        def testMeta = metaCtor.newInstance(testTags, "Backup Verification Document", null, 4, null)

                        def alcyoneVfsCompanion = classLoader.loadClass("com.github.noahshen.alcyone.context.vfs.runtime.AlcyoneVfs").getField("Companion").get(null)
                        def writeOptionsClass = classLoader.loadClass("com.github.noahshen.alcyone.context.vfs.WriteOptions")
                        def writeModeClass = classLoader.loadClass("com.github.noahshen.alcyone.context.vfs.WriteMode")
                        def createNewMode = Enum.valueOf(writeModeClass, "CREATE_NEW")
                        def writeOptions = writeOptionsClass.getConstructor(writeModeClass).newInstance(createNewMode)

                        def statOptionsClass = classLoader.loadClass("com.github.noahshen.alcyone.context.vfs.StatOptions")
                        def statOptions = statOptionsClass.getConstructor().newInstance()

                        def readOptionsClass = classLoader.loadClass("com.github.noahshen.alcyone.context.vfs.ReadOptions")
                        def readOptions = readOptionsClass.getConstructors().find { it.parameterCount == 3 }.newInstance(null, 1, null)

                        def buildersKt = classLoader.loadClass("kotlinx.coroutines.BuildersKt")
                        def emptyContext = classLoader.loadClass("kotlin.coroutines.EmptyCoroutineContext").INSTANCE
                        def function2Class = classLoader.loadClass("kotlin.jvm.functions.Function2")
                        def runBlockingMethod = buildersKt.getMethod("runBlocking", classLoader.loadClass("kotlin.coroutines.CoroutineContext"), function2Class)

                        def runBlocking = { Closure cl ->
                            def proxy = java.lang.reflect.Proxy.newProxyInstance(
                                classLoader,
                                [function2Class] as Class[],
                                new java.lang.reflect.InvocationHandler() {
                                    Object invoke(Object proxyObj, java.lang.reflect.Method method, Object[] args) {
                                        if (method.name == "invoke") {
                                            return cl.call(args[0], args[1])
                                        }
                                        return method.invoke(this, args)
                                    }
                                }
                            )
                            return runBlockingMethod.invoke(null, emptyContext, proxy)
                        }

                        println("Step 1: Create VFS instance and write data...")
                        def vfs1 = runBlocking { scope, cont ->
                            alcyoneVfsCompanion.create(config, cont)
                        }

                        def info = runBlocking { scope, cont ->
                            vfs1.write(testUri, testContent, writeOptions, cont)
                        }
                        def originalNodeId = info.id
                        println("  Written file nodeId: $originalNodeId")

                        runBlocking { scope, cont ->
                            vfs1.setMetadata(originalNodeId, testMeta, cont)
                        }
                        println("  Set metadata: tags=${testMeta.tags}, desc=${testMeta.description}")

                        println("Step 2: Normal close...")
                        runBlocking { scope, cont ->
                            vfs1.close(cont)
                        }
                        println("  VFS instance successfully closed.")

                        println("Step 3: Check SQLite journal_mode & file layout...")
                        registeredDriver.connect("jdbc:sqlite:" + stateDb.toAbsolutePath(), new Properties()).withCloseable { conn ->
                            conn.createStatement().withCloseable { stmt ->
                                stmt.executeQuery("PRAGMA journal_mode;").withCloseable { rs ->
                                    if (rs.next()) {
                                        def journalMode = rs.getString(1)
                                        println("  PRAGMA journal_mode: " + journalMode)
                                        assert "delete".equalsIgnoreCase(journalMode)
                                    }
                                }
                            }
                        }

                        def filesInTempDir = java.nio.file.Files.list(tempDir).map { it.fileName.toString() }.toList().sort()
                        println("  Directory files: $filesInTempDir")
                        assert filesInTempDir.contains("state.db")
                        assert filesInTempDir.contains("state.db.lock")
                        assert filesInTempDir.contains("docs")
                        assert !filesInTempDir.contains("state.db-wal")
                        assert !filesInTempDir.contains("state.db-shm")
                        assert !filesInTempDir.contains("state.db-journal")
                        println("  File layout check passed (SQLite data file: state.db; lock file: state.db.lock; data dir: docs; no wal/shm/journal).")

                        println("Step 4: Backup state.db and physical files to backup directory...")
                        def backupDb = backupDir.resolve("state.db")
                        java.nio.file.Files.copy(stateDb, backupDb)
                        def backupDocs = backupDir.resolve("docs")
                        java.nio.file.Files.createDirectories(backupDocs)
                        java.nio.file.Files.walk(docsDir).withCloseable { stream ->
                            stream.forEach { source ->
                                def dest = backupDocs.resolve(docsDir.relativize(source))
                                if (java.nio.file.Files.isDirectory(source)) {
                                    if (!java.nio.file.Files.exists(dest)) java.nio.file.Files.createDirectories(dest)
                                } else {
                                    java.nio.file.Files.copy(source, dest)
                                }
                            }
                        }
                        println("  Backup completed: state.db and docs copied.")

                        println("Step 5: Completely purge original directory...")
                        java.nio.file.Files.walk(tempDir).withCloseable { stream ->
                            def toDelete = stream.sorted(Comparator.reverseOrder()).toList()
                            toDelete.each { targetPath ->
                                if (targetPath != tempDir) java.nio.file.Files.delete(targetPath)
                            }
                        }
                        def emptyCheck = java.nio.file.Files.list(tempDir).map { it.fileName.toString() }.toList()
                        println("  Original dir contents after purge: $emptyCheck")
                        assert emptyCheck.isEmpty()

                        println("Step 6: Restore from backup...")
                        java.nio.file.Files.copy(backupDb, stateDb)
                        java.nio.file.Files.createDirectories(docsDir)
                        java.nio.file.Files.walk(backupDocs).withCloseable { stream ->
                            stream.forEach { source ->
                                def dest = docsDir.resolve(backupDocs.relativize(source))
                                if (java.nio.file.Files.isDirectory(source)) {
                                    if (!java.nio.file.Files.exists(dest)) java.nio.file.Files.createDirectories(dest)
                                } else {
                                    java.nio.file.Files.copy(source, dest)
                                }
                            }
                        }
                        println("  Restore completed.")

                        println("Step 7: Start new VFS instance with same config & verify assertions...")
                        def vfs2 = runBlocking { scope, cont ->
                            alcyoneVfsCompanion.create(config, cont)
                        }

                        try {
                            def restoredInfo = runBlocking { scope, cont ->
                                vfs2.stat(testUri, statOptions, cont)
                            }
                            println("  stat(uri).id: " + restoredInfo.id + " (expected: " + originalNodeId + ")")
                            assert originalNodeId == restoredInfo.id

                            def restoredContent = runBlocking { scope, cont ->
                                vfs2.read(testUri, readOptions, cont)
                            }
                            println("  read(uri) length: " + restoredContent.length + " bytes")
                            assert Arrays.equals(testContent, restoredContent)

                            def restoredMeta = runBlocking { scope, cont ->
                                vfs2.getMetadata(originalNodeId, cont)
                            }
                            println("  getMetadata(nodeId): tags=" + restoredMeta.tags + ", desc=" + restoredMeta.description)
                            assert testTags.equals(restoredMeta.tags)
                            assert testMeta.description.equals(restoredMeta.description)

                            println("  ALL ASSERTIONS PASSED!")
                        } finally {
                            runBlocking { scope, cont ->
                                vfs2.close(cont)
                            }
                            println("  Restored VFS instance closed.")
                        }

                        println("=== Probe Execution Completed Successfully ===")
                    } finally {
                        def deleteRecursively = { java.nio.file.Path targetDir ->
                            if (java.nio.file.Files.exists(targetDir)) {
                                java.nio.file.Files.walk(targetDir).withCloseable { stream ->
                                    stream.sorted(Comparator.reverseOrder()).forEach { targetFile ->
                                        try { java.nio.file.Files.deleteIfExists(targetFile) } catch (Exception ignored) {}
                                    }
                                }
                            }
                        }
                        deleteRecursively(tempDir)
                        deleteRecursively(backupDir)
                    }
                }
            }
        }
    }
}
```
